\section*{Part A. Renewable sources and Study 1 microgrid (proposal Sections 3.3--3.5.1)}
% PV cell temperature
\begin{equation}
T_{\mathrm{cell}} = T_{\mathrm{amb}} + k_{\mathrm{NOCT}}\,G \tag{3.1}
\end{equation}
% PV output power
\begin{equation}
P_{\mathrm{pv}} = \eta_{\mathrm{deg}}\,P_{\mathrm{pv,rated}}\,\frac{G}{G_{\mathrm{ref}}}\left[1 - C_T\left(T_{\mathrm{cell}} - T_{\mathrm{ref}}\right)\right], \qquad 0 \le P_{\mathrm{pv}} \le P_{\mathrm{pv,rated}} \tag{3.2}
\end{equation}
% Wind turbine mechanical power (Study 1)
\begin{equation}
P_m = \frac{1}{2}\,\rho\,\pi R^{2}\,C_p(\lambda,\beta)\,v_w^{3} \tag{3.3}
\end{equation}
% Power coefficient of the turbine
\begin{equation}
C_p = 0.5176\left(\frac{116}{\lambda_i} - 0.4\,\beta - 5\right)e^{-21/\lambda_i} + 0.0068\,\lambda, \qquad \frac{1}{\lambda_i} = \frac{1}{\lambda + 0.08\,\beta} - \frac{0.035}{\beta^{3} + 1} \tag{3.4}
\end{equation}
% Wind power curve with density correction (Study 2)
\begin{equation}
P_w = \begin{cases} P_r\,\dfrac{v'^{3} - v_{\mathrm{ci}}^{3}}{v_r^{3} - v_{\mathrm{ci}}^{3}}, & v_{\mathrm{ci}} \le v' < v_r \\ P_r, & v_r \le v' < v_{\mathrm{co}} \\ 0, & \text{otherwise} \end{cases} \qquad v' = v\left(\frac{\rho}{1.225}\right)^{1/3} \tag{3.5}
\end{equation}
% VSG power injection (Study 1)
\begin{equation}
p_{\mathrm{vsg}} = -2\,J_{\mathrm{eff}}\,\frac{d\Delta\omega}{dt} - D_{\mathrm{eff}}\,\Delta\omega_m \tag{3.6}
\end{equation}
% Closed-loop swing equation (Study 1)
\begin{equation}
2\left(H_{\mathrm{sys}} + J_{\mathrm{eff}}\right)\frac{d\Delta\omega}{dt} = \Delta P_{\mathrm{ext}} - D_{\mathrm{eff}}\,\Delta\omega_m \tag{3.7}
\end{equation}
% Power imbalance and BESS droop (Study 1)
\begin{equation}
\Delta P_{\mathrm{ext}} = \frac{P_{\mathrm{pv}} + P_w + P_{\mathrm{sch}} - P_{\mathrm{load}}}{S_{\mathrm{base}}} + p_{\mathrm{dr}}, \qquad T_{\mathrm{dr}}\,\frac{dp_{\mathrm{dr}}}{dt} = -\frac{1}{R}\,\Delta\omega_m - p_{\mathrm{dr}} \tag{3.8}
\end{equation}
% BESS scheduling and voltage-dependent load (Study 1)
\begin{equation}
\tau_{\mathrm{sch}}\,\frac{dP_{\mathrm{gen},f}}{dt} = P_{\mathrm{pv}} + P_w - P_{\mathrm{gen},f}, \qquad P_{\mathrm{sch}} = P_{\mathrm{load,nom}}\left(\frac{V}{V_{\mathrm{nom}}}\right)^{k_p} - P_{\mathrm{gen},f} \tag{3.9}
\end{equation}
% Reactive-power droop with apparent-power limit (Study 1)
\begin{equation}
q_{\mathrm{vsg}} = K_q\left(V_{\mathrm{nom}} - V\right), \qquad \left|q_{\mathrm{vsg}}\right| \le \frac{\sqrt{S_{\mathrm{rated}}^{2} - P_{\mathrm{bess}}^{2}}}{S_{\mathrm{base}}} \tag{3.10}
\end{equation}
% PLL frequency measurement (Study 1)
\begin{equation}
\tau_{\mathrm{pll}}\,\frac{df_m}{dt} = f + n - f_m, \qquad n \sim \mathcal{N}\!\left(0,\ \sigma_{\mathrm{pll}}^{2}\right) \tag{3.11}
\end{equation}
\section*{Part B. Study 2 hybrid diesel--VSG microgrid (proposal Section 3.5.2, IREA Section III.A)}
% VSG electrical power and net imbalance  [IREA (1)]
\begin{equation}
P_v = \frac{K_s\left(K_d\,\delta + \Delta P_{\mathrm{net}}\right)}{K_d + K_s}, \qquad \Delta P_{\mathrm{net}} = \Delta P_L - \Delta P_w + D_L\,P_{L0}\,\Delta\omega_b \tag{3.12}
\end{equation}
% Bus frequency  [IREA (2)]
\begin{equation}
\Delta\omega_b = \frac{K_d\,\Delta\omega_d + K_s\,\Delta\omega_v}{K_d + K_s} \tag{3.13}
\end{equation}
% Diesel generator swing equation  [IREA (3)]
\begin{equation}
2H_d\,\frac{d\Delta\omega_d}{dt} = \Delta P_m - \Delta P_{e,d} - D_d\left(\Delta\omega_d - \Delta\omega_b\right) \tag{3.14}
\end{equation}
% VSG virtual-rotor equations  [IREA (4)]
\begin{equation}
2J_v\,\frac{d\Delta\omega_v}{dt} = P_{\mathrm{set}} - P_v - D_v\,\Delta\omega_v, \qquad \frac{d\delta}{dt} = \omega_0\left(\Delta\omega_v - \Delta\omega_d\right) \tag{3.15}
\end{equation}
% VSG power band  [IREA (5)]
\begin{equation}
-\left(P_{\mathrm{ch,max}}(\mathrm{SoC}) + P_{b0}\right) + P_{\mathrm{pv},s} \;\le\; P_v \;\le\; P_{\mathrm{dis,max}}(\mathrm{SoC}) - P_{b0} + P_{\mathrm{pv},s} \tag{3.16}
\end{equation}
% PV share of the support
\begin{equation}
T_{\mathrm{pv}}\,\frac{dP_{\mathrm{pv},s}}{dt} = \mathrm{clip}\!\left(\alpha\,P_{v,\mathrm{req}},\ -P_{\mathrm{pv},0},\ d\,P_{\mathrm{MPP}}\right) - P_{\mathrm{pv},s} \tag{3.16a}
\end{equation}
\section*{Part C. Converter-headroom feasibility (proposal Section 3.6, IREA Section V)}
% Summed headroom requirement (Study 1)
\begin{equation}
h_{\mathrm{req}} = 2J\,\frac{\mathrm{RoCoF}_{\max}}{f_0} + D\,\frac{\Delta f_{\max}}{f_0} = 0.08\,J + 0.01\,D \tag{3.17}
\end{equation}
% Summed headroom requirement (Study 2 notation)  [IREA (7)]
\begin{equation}
h_{\mathrm{req,sum}} = 2J_v\,\mathrm{RoCoF}_{\mathrm{lim}} + D_v\,\Delta f_{\mathrm{lim}} \tag{IREA 7}
\end{equation}
% Available headroom (Study 1)
\begin{equation}
h_{\mathrm{avail}} = \frac{S_{\mathrm{rated}} - \left|P_{\mathrm{sch}} + p_{\mathrm{dr}}\,S_{\mathrm{base}}\right|}{S_{\mathrm{base}}} \tag{3.18}
\end{equation}
% Proportional derating  [IREA (9)]
\begin{equation}
\left(J_{\mathrm{eff}},\,D_{\mathrm{eff}}\right) = \left(J,\,D\right)\cdot\min\!\left(1,\ \frac{h_{\mathrm{avail}}}{h_{\mathrm{req}}}\right) \tag{3.19}
\end{equation}
% Direction-aware headroom (Study 2)  [IREA (6)]
\begin{equation}
h_{\mathrm{up}} = \max\!\left(P_{\mathrm{dis,max}}(\mathrm{SoC}) - P_{b0},\ 0\right) + d\,P_{\mathrm{MPP}}, \qquad h_{\mathrm{dn}} = \max\!\left(P_{\mathrm{ch,max}}(\mathrm{SoC}) + P_{b0},\ 0\right) + P_{\mathrm{pv},0} \tag{3.20}
\end{equation}
% Time-separated (max) headroom requirement  [IREA (8)]
\begin{equation}
h_{\mathrm{req,max}} = \max\!\left(2J_v\,\mathrm{RoCoF}_{\mathrm{lim}},\ D_v\,\Delta f_{\mathrm{band}}\right) \le h \tag{3.21}
\end{equation}
\section*{Part D. Learning formulation (proposal Sections 3.7--3.8, IREA Section VI)}
% Residual action mapping (Study 2)
\begin{equation}
J_v = \begin{cases} J_0 + a_1\left(J_{\mathrm{ub}} - J_0\right), & a_1 \ge 0 \\ J_0 + a_1\left(J_0 - J_{\min}\right), & a_1 < 0 \end{cases} \tag{3.22}
\end{equation}
% Interval cost (Study 2)  [IREA (10)]
\begin{equation}
c = w_f\,\frac{\left\langle\left(|\Delta f| - 0.2\right)_{+}^{2}\right\rangle}{0.5^{2}} + w_{\mathrm{in}}\,\frac{\left\langle\Delta f^{2}\right\rangle}{0.5^{2}} + w_r\left\langle\left(|\mathrm{RoCoF}| - 1\right)_{+}^{2}\right\rangle + w_{\mathrm{fast}}\,\frac{\left\langle\left(P_v - P_{\mathrm{set}}\right)^{2}\right\rangle}{P_{b,\max}^{2}} + w_b\,\frac{\left\langle\Delta P_b^{2}\right\rangle}{P_{b,\max}^{2}} + w_s\,\frac{\langle\mathrm{sat}\rangle}{0.05} + w_v\,\mathrm{viol} \tag{3.23}
\end{equation}
% TD3 critic target
\begin{equation}
y = r + \gamma \min_{i=1,2} Q_{\phi'_i}\!\left(s',\ \mu_{\theta'}(s') + \varepsilon\right), \qquad \varepsilon \sim \mathrm{clip}\!\left(\mathcal{N}(0,\sigma^{2}),\ -c,\ c\right) \tag{3.24}
\end{equation}
\section*{Part E. Metrics and oracle bound (proposal Section 3.9, IREA Section VI.B)}
% Frequency error integrals
\begin{equation}
\mathrm{IAE} = \int_0^{T} |\Delta f|\,dt, \qquad \mathrm{ISE} = \int_0^{T} \Delta f^{2}\,dt, \qquad \mathrm{ITAE} = \int_0^{T} t\,|\Delta f|\,dt \tag{3.25}
\end{equation}
% Oracle gain
\begin{equation}
G_{\mathrm{oracle}} = \frac{1}{N}\sum_{i=1}^{N}\max_{k} R_{ik} - \max_{k}\,\frac{1}{N}\sum_{i=1}^{N} R_{ik} \tag{3.26}
\end{equation}
